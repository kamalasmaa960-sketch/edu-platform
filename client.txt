function Assignment(c, d) {
  const a = d.assignment, box = h("div", { class: "box" }), ans = {};
  const draw = (res) => {
    box.replaceChildren(h("h2", {}, a.title), res ? h("div", { class: "okm" }, `نتيجتك: ${res.score} من ${res.total}`) : null);
    a.questions.forEach((q, qi) => {
      const dt = res && res.details.find((x) => x.id === q.id);
      box.append(h("div", {}, h("b", {}, `${qi + 1}. `), rich(q.q), q.opts.map((o, i) => h("button", { type: "button", class: "opt" + (dt ? (i === dt.correct ? " ok" : i === dt.chosen ? " bad" : "") : ans[q.id] === i ? " sel" : ""), disabled: !!res, onclick: () => { ans[q.id] = i; draw(); } }, rich(o)))));
    });
    if (!res) box.append(h("button", { class: "btn", onclick: async (e) => { if (Object.keys(ans).length < a.questions.length && !confirm("لم تجب عن كل الأسئلة. إرسال؟")) return; e.target.disabled = true; try { const r = await api("assignment/" + c.id + "/submit", "POST", { answers: ans }); draw(r.result); } catch (x) { box.append(msg(x.message)); e.target.disabled = false; } } }, "تسليم الواجب"));
    else box.append(h("button", { class: "btn o", onclick: () => { for (const k in ans) delete ans[k]; draw(); } }, "إعادة المحاولة"));
  };
  draw(d.result); return box;
}
